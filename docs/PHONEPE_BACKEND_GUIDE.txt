 --- 4f) PaymentController - createOrder (sabse important) -------------------
app/Http/Controllers/PaymentController.php

<?php
namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\DB;
use App\Models\Payment;

class PaymentController extends Controller
{
    private function generateOrderId(): string {
        return 'GMT' . rand(10,100) . date('Ymd') . rand(100,1000) . time();
    }

    public function createOrder(Request $request)
    {
        $amount = (float) $request->input('amount'); // INR
        if ($amount <= 0) {
            return response()->json(['status' => false, 'message' => 'Invalid amount'], 400);
        }

        $merchantId  = config('phonepe.merchant_id');
        $saltKey     = config('phonepe.salt_key');
        $saltIndex   = config('phonepe.salt_index');
        $baseUrl     = config('phonepe.base_url');
        $redirectUrl = config('phonepe.redirect_url');

        $orderId = $this->generateOrderId();
        $userId  = $request->user()->id ?? 'GUEST';

        $payload = [
            'merchantId'            => $merchantId,
            'merchantTransactionId' => $orderId,
            'merchantUserId'        => 'MUID' . $userId,
            'amount'                => (int) round($amount * 100), // PAISE
            'redirectUrl'           => $redirectUrl,
            'redirectMode'          => 'POST',
            'callbackUrl'           => $redirectUrl,
            'merchantOrderId'       => $orderId,
            'mobileNumber'          => $request->input('mobile', '9999999999'),
            'message'               => 'Payment for Gaumata Sewa Trust',
            'paymentInstrument'     => ['type' => 'PAY_PAGE'],
        ];

        $base64Payload = base64_encode(json_encode($payload));

        // checksum: base64payload + "/pg/v1/pay" + saltKey  -> sha256 ### saltIndex
        $checksum = hash('sha256', $base64Payload . '/pg/v1/pay' . $saltKey) . '###' . $saltIndex;

        $response = Http::withHeaders([
            'Content-Type' => 'application/json',
            'X-VERIFY'     => $checksum,
            'accept'       => 'application/json',
        ])->post($baseUrl . '/pg/v1/pay', ['request' => $base64Payload]);

        $data = $response->json();

        if (($data['success'] ?? false) == '1' && !empty($data['data']['token'])) {
            // DB mein pending order record save kar lo
            Payment::create([
                'user_id' => is_numeric($userId) ? $userId : null,
                'order_id' => $orderId,
                'amount'   => (int) round($amount * 100),
                'status'   => 'PENDING',
            ]);

            return response()->json([
                'status'  => true,
                'orderId' => $orderId,
                'token'   => $data['data']['token'],
            ]);
        }

        return response()->json([
            'status'  => false,
            'message' => $data['message'] ?? 'Order creation failed',
            'code'    => $data['code'] ?? null,
        ], 500);
    }

    // comment: sandbox test ke liye URL change karo:
    // https://api-preprod.phonepe.com/apis/pg-sandbox
}


 --- 4g) PaymentController - callback (webhook, authoritative confirm) --------
    public function callback(Request $request)
    {
        // Step A: X-VERIFY header se confirm karo ki request PhonePe se hai
        $saltKey   = config('phonepe.salt_key');
        $saltIndex = config('phonepe.salt_index');
        $incoming  = $request->header('X-VERIFY');
        $body      = $request->getContent();

        $expected = hash('sha256', $body . $saltKey) . '###' . $saltIndex;
        if ($incoming !== $expected) {
            return response()->json(['status' => false], 401);
        }

        // Step B: response parse karo
        $payload = $request->json('response');
        $orderId = $payload['merchantTransactionId'] ?? null;
        $state   = $payload['state'] ?? 'PENDING';   // COMPLETED | FAILED | PENDING
        $ppTxn   = $payload['transactionId'] ?? null;

        $payment = Payment::where('order_id', $orderId)->first();
        if (!$payment) {
            return response()->json(['status' => false], 404);
        }

        if ($state === 'COMPLETED') {
            $payment->status = 'SUCCESS';
            $payment->phonepe_transaction_id = $ppTxn;
            $payment->response_payload = $payload;
            $payment->save();

            // YAHAN activation logic likho:
            //  - agar plan purchase hai -> user ko membership/plan assign karo
            //  - agar donation hai       -> donation record banao
            //  - user ko SUCCESS receipt/navigate
        } elseif ($state === 'FAILED') {
            $payment->status = 'FAILED';
            $payment->response_payload = $payload;
            $payment->save();
        }

        return response()->json(['status' => true]);
    }


 --- 4h) PaymentController - status verify ------------------------------------
    // Frontend SDK jab SUCCESS return karta hai, toh bhi backend se confirm
    // karna best practice hai (webhook general aata hai). Ye endpoint optional hai.
    public function status($orderId)
    {
        $merchantId = config('phonepe.merchant_id');
        $saltKey    = config('phonepe.salt_key');
        $saltIndex  = config('phonepe.salt_index');
        $baseUrl    = config('phonepe.base_url');

        $path = '/pg/v1/status/' . $merchantId . '/' . $orderId;
        $checksum = hash('sha256', $path . $saltKey) . '###' . $saltIndex;

        $response = Http::withHeaders([
            'Content-Type'   => 'application/json',
            'X-VERIFY'       => $checksum,
            'X-MERCHANT-ID'  => $merchantId,
        ])->get($baseUrl . $path);

        $data = $response->json()['data'] ?? null;
        return response()->json([
            'status' => true,
            'payment_status' => $data['state'] ?? 'UNKNOWN',
            'transaction_id' => $data['transactionId'] ?? null,
            'response'       => $data,
        ]);
    }
}

--------------------------------------------------------------------------------
 8) TROUBLESHOOTING (common issues)
--------------------------------------------------------------------------------
[1] init error "expected argument count: 4"
    -> RN side fixed hai (positional args). Backend se farak nahi padta.

[2] createOrder error / "Order creation failed"
    -> Backend ka PhonePe call reject. Response code/message dekho.
       Most common: wrong API_KEY / wrong URL / amount format.

[3] SDK SUCCESS milta hai par payment actually fail
    -> SDK ka SUCCESS matlab PhonePe app flow complete. AUTHORITATIVE confirm
       webhook (callback) / status API se hai. Donation tabhi activate karo jab
       callback COMPLETED aaye. (Don't trust only SDK status for real money.)

[4] Currency/amount mismatch
    -> Frontend RUPEES bhejta hai (e.g. 100). Backend PAISE bhejta hai (10000).
       Phir bhi koi confusion ho toh request payload check karo.

[5] Webhook 401 Unauthorized
    -> X-VERIFY mismatch. Salt key/salt index .env wise confirm karo. Same salt
       index jo order create mein use hua.

[6] Token empty / null
    -> PhonePe ne pay rejected. Response ka code+message dekho. (e.g. 8101 =
       transaction already exists, 8070 = session expired etc.)
